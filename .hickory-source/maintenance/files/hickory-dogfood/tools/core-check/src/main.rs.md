//! Verify the complete construction with the reproduced Hickory engine itself.
use hick_lang::{EntryKind, ProjectTree, evaluate_project, load_project_sources};
use sha2::{Digest, Sha256};
use std::{collections::BTreeMap, path::PathBuf};

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let mut args = std::env::args().skip(1);
    let root = PathBuf::from(args.next().ok_or("expected project root")?);
    let baseline = PathBuf::from(args.next().ok_or("expected baseline JSON")?);
    let materialize = match args.next().as_deref() { None => false, Some("--materialize") => true, _ => return Err("unexpected arguments".into()) };
    if args.next().is_some() { return Err("unexpected arguments".into()); }
    let sources = load_project_sources("project.md", &|name| {
        std::fs::read_to_string(root.join(name)).map_err(|e| e.to_string())
    })?;
    let generated = evaluate_project("project.md", &sources, &ProjectTree::default())?;
    let baseline: serde_json::Value = serde_json::from_slice(&std::fs::read(baseline)?)?;
    let expected: BTreeMap<_, _> = baseline["entries"].as_array().ok_or("missing entries")?
        .iter().map(|entry| (entry["path"].as_str().unwrap(), entry)).collect();
    if expected.len() != generated.new.entries().len() {
        return Err("complete generated inventory differs from captured inventory".into());
    }
    for (path, entry) in generated.new.entries() {
        let captured = expected.get(path.as_str()).ok_or("unexpected generated path")?;
        let kind = if entry.kind == EntryKind::File { "file" } else { "symlink" };
        if captured["kind"] != kind || captured["executable"] != entry.executable
            || captured["bytes"] != entry.bytes.len()
            || captured["sha256"] != format!("{:x}", Sha256::digest(&entry.bytes)) {
            return Err(format!("generated bytes or metadata differ: {path}").into());
        }
    }
    if materialize { hick_project_env::write_tree(&root, &generated.new)?; }
    println!("Reproduced hick-lang engine verified {} entries from {} Markdown sources against the independent capture.", expected.len(), sources.len());
    Ok(())
}
