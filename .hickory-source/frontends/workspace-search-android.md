# Search saved text in a floating Android view

For maintainers changing workspace services independently of the editor renderer.

Search results are snapshots of saved workspace text. Keep every occurrence and its byte column, including generated-file ownership. Convert a byte column only at a Unicode scalar boundary; editors use UTF-16, so a character before the match may occupy two units. A result may reveal a precise position only after the open buffer matches the reported source hash. No search edits or saves source. Query state owns request identities, not the presentation. Editing options or reconnecting invalidates pending reads. A submitted request publishes only into its original query state. The existing Edit menu presents this view without a toolbar or a new docked pane.

package com.hickorydocs.simple
import androidx.compose.runtime.*
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.rememberScrollState
import androidx.compose.foundation.verticalScroll
import androidx.compose.material3.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.launch
import java.util.UUID
class WorkspaceSearch {
    var query by mutableStateOf(""); private set
    var regex by mutableStateOf(false); private set
    var matchCase by mutableStateOf(false); private set
    var wholeWord by mutableStateOf(false); private set
    var result by mutableStateOf<WorkspaceSearchResult?>(null)
    var notice by mutableStateOf("")
    var page by mutableStateOf(0)
    var revision = UUID.randomUUID()
    fun invalidate() { revision = UUID.randomUUID(); result = null; notice = ""; page = 0 }
    fun updateQuery(value: String) { query = value; invalidate() }
    fun updateOptions(regex: Boolean = this.regex, matchCase: Boolean = this.matchCase, wholeWord: Boolean = this.wholeWord) {
        this.regex = regex; this.matchCase = matchCase; this.wholeWord = wholeWord; invalidate()
    }
    suspend fun run(engine: Engine?) {
        if (engine == null || query.isEmpty()) { notice = "Connect to a workspace and enter a query."; return }
        revision = UUID.randomUUID(); val request = revision; result = null; page = 0; notice = "Searching saved workspace text…"
        try { val answer = engine.find(query, regex, matchCase, wholeWord); if (revision == request) { result = answer; notice = "" } }
        catch (cancelled: CancellationException) { throw cancelled }
        catch (failure: Exception) { if (revision == request) notice = "${failure.message} Check the query and retry." }
    }
}
@Composable fun WorkspaceSearchView(workspace: Workspace) {
    val search = workspace.search; val scope = rememberCoroutineScope()
    val hits = search.result?.hits.orEmpty(); val last = maxOf(0, (hits.size + 99) / 100 - 1); val selected = minOf(search.page, last)
    Column(Modifier.widthIn(min = 260.dp, max = 380.dp).heightIn(max = 440.dp).verticalScroll(rememberScrollState()).padding(8.dp)) {
        Text("Find in workspace", fontSize = 16.sp)
        Text("Saved workspace text · unsaved edits are preserved", fontSize = 11.sp)
        OutlinedTextField(search.query, { search.updateQuery(it) }, label = { Text("Workspace search query") }, singleLine = true)
        Row { Checkbox(search.regex, { search.updateOptions(regex = it) }); Text("Regular expression") }
        Row { Checkbox(search.matchCase, { search.updateOptions(matchCase = it) }); Text("Match case") }
        Row { Checkbox(search.wholeWord, { search.updateOptions(wholeWord = it) }); Text("Whole word") }
        TextButton(enabled = search.query.isNotEmpty(), onClick = { scope.launch { search.run(workspace.engine) } }) { Text("Search") }
        if (search.notice.isNotEmpty()) Text(search.notice, fontSize = 12.sp)
        search.result?.let { Text("${hits.size} occurrences" + if (it.truncated) " · Result cap reached; narrow the query" else "", fontSize = 12.sp) }
        for (hit in hits.drop(selected * 100).take(100)) {
            TextButton(onClick = { scope.launch { workspace.openSearchHit(hit) } }) { Text("${hit.path}:${hit.line} · ${hit.text}", fontSize = 12.sp) }
            hit.generatedBy?.let { Text("Generated by $it · read-only", fontSize = 11.sp) }
        }
        if (last > 0) Row { TextButton(enabled = selected > 0, onClick = { search.page = selected - 1 }) { Text("Previous results") }; Text("${selected + 1}/${last + 1}"); TextButton(enabled = selected < last, onClick = { search.page = selected + 1 }) { Text("Next results") } }
    }
}
