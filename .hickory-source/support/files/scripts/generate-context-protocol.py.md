#!/usr/bin/env python3
"""Generate wire types and packaged schema; --check detects drift without builds."""
import json
from pathlib import Path
import re
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[1]
SCHEMA = ROOT / 'protocols/context-v1/schema.json'
schema = json.loads(SCHEMA.read_text())

def variant(value):
    return ''.join(word.capitalize() for word in re.split(r'[_.]', value))

def rust_type(spec):
    if '$ref' in spec:
        return spec['$ref'].split('/')[-1]
    return {'string': 'String', 'integer': 'u64', 'number': 'f64',
            'boolean': 'bool', 'object': 'serde_json::Value'}.get(spec['type']) or f'Vec<{rust_type(spec["items"])}>'

def py_type(spec):
    if '$ref' in spec:
        return spec['$ref'].split('/')[-1]
    if 'const' in spec:
        return f'Literal[{spec["const"]!r}]'
    return {'string': 'str', 'integer': 'int', 'number': 'float',
            'boolean': 'bool', 'object': 'dict[str, object]'}.get(spec['type']) or f'list[{py_type(spec["items"])}]'

rust = ['// Generated by scripts/generate-context-protocol.py; do not edit.',
        'use serde::{Deserialize, Serialize};']
python = ['# Generated by scripts/generate-context-protocol.py; do not edit.',
          'from __future__ import annotations',
          'from typing import Literal, NotRequired, TypedDict']
for name, spec in schema['$defs'].items():
    if 'enum' in spec:
        rust += ['#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]', f'pub enum {name} {{']
        for value in spec['enum']:
            rust += [f'    #[serde(rename = "{value}")]', f'    {variant(value)},']
        rust += ['}']
        python += [f'{name} = Literal[{", ".join(repr(x) for x in spec["enum"])}]']
    else:
        rust += ['#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]', '#[serde(deny_unknown_fields)]', f'pub struct {name} {{']
        python += [f'class {name}(TypedDict):']
        for field, prop in spec['properties'].items():
            rt, pt = rust_type(prop), py_type(prop)
            if field not in spec['required']:
                rt, pt = f'Option<{rt}>', f'NotRequired[{pt}]'
                rust += ['    #[serde(default, skip_serializing_if = "Option::is_none")]']
            rust += [f'    pub {field}: {rt},']
            python += [f'    {field}: {pt}']
        rust += ['}']
    python += ['']
rust += ['pub(crate) fn deserializes(name: &str, value: &serde_json::Value) -> bool {', 'match name {']
for name in schema['$defs']:
    rust += [f'\"{name}\" => serde_json::from_value::<{name}>(value.clone()).is_ok(),']
rust += ['_ => false,', '}', '}']
rust = subprocess.run(['rustfmt', '--edition', '2024'], input='\n'.join(rust)+'\n', text=True, capture_output=True, check=True).stdout
outputs = {
    ROOT / 'crates/hick-context-protocol/src/types.rs': rust,
    ROOT / 'apps/agent/src/hickory_agent_worker/context_types.py': '\n'.join(python),
    ROOT / 'apps/agent/src/hickory_agent_worker/context_schema.json': SCHEMA.read_text(),
}
for path, content in outputs.items():
    if '--check' in sys.argv:
        if not path.exists() or path.read_text() != content:
            raise SystemExit(f'Stale generated context protocol: {path.relative_to(ROOT)}')
    else:
        path.write_text(content)
