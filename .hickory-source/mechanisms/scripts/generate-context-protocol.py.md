def rust_type(spec):
    if '$ref' in spec:
        return spec['$ref'].split('/')[-1]
    return {'string': 'String', 'integer': 'u64', 'number': 'f64',
            'boolean': 'bool', 'object': 'serde_json::Value'}.get(spec['type']) or f'Vec<{rust_type(spec["items"])}>'

def py_type(spec):
    if '$ref' in spec:
        return spec['$ref'].split('/')[-1]
    if 'const' in spec:
        return f'Literal[{spec["const"]!r}]'
    return {'string': 'str', 'integer': 'int', 'number': 'float',
            'boolean': 'bool', 'object': 'dict[str, object]'}.get(spec['type']) or f'list[{py_type(spec["items"])}]'

for name, spec in schema['$defs'].items():
    if 'enum' in spec:
        rust += ['#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]', f'pub enum {name} {{']
        for value in spec['enum']:
            rust += [f'    #[serde(rename = "{value}")]', f'    {variant(value)},']
        rust += ['}']
        python += [f'{name} = Literal[{", ".join(repr(x) for x in spec["enum"])}]']
    else:
        rust += ['#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]', '#[serde(deny_unknown_fields)]', f'pub struct {name} {{']
        python += [f'class {name}(TypedDict):']
        for field, prop in spec['properties'].items():
            rt, pt = rust_type(prop), py_type(prop)
            if field not in spec['required']:
                rt, pt = f'Option<{rt}>', f'NotRequired[{pt}]'
                rust += ['    #[serde(default, skip_serializing_if = "Option::is_none")]']
            rust += [f'    pub {field}: {rt},']
            python += [f'    {field}: {pt}']
        rust += ['}']
    python += ['']
