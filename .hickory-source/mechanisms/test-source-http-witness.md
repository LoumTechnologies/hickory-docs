# Exercise source test admission through HTTP

For maintainers changing the source-to-run boundary.

Start an isolated engine and use its public HTTP boundary. The fixture helper only manages process lifetime; no browser or broad test suite is started.

<ui:copy id="source-tests-http-fixture">
"""Focused source-test protocol acceptance against an isolated, real engine."""
import importlib.util
import json
import os
from pathlib import Path
import tempfile
import time
import urllib.error
import urllib.request

ROOT = Path(__file__).resolve().parents[1]


def main():
    spec = importlib.util.spec_from_file_location("simple_fixture", ROOT / "scripts/test-simple-browser.py")
    fixture = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(fixture)
    with tempfile.TemporaryDirectory(prefix="hickory-source-tests-") as directory:
        root = Path(directory)
        environment = dict(os.environ, HICKORY_STATE_DIR=str(root / "state"))
        engine = fixture.start([os.environ.get("HICKORY_SIMPLE_TEST_ENGINE", str(ROOT / "target/source-edit-focus/debug/examples/engine_client")), str(root), "source-tests"], env=environment)
        try:
            base = fixture.origin(engine)

            def call(path, body=None, status=200):
                request = urllib.request.Request(base + "/api/" + path, data=None if body is None else json.dumps(body).encode(), headers={"Content-Type": "application/json"})
                try:
                    with urllib.request.urlopen(request, timeout=20) as response:
                        actual, data = response.status, response.read()
                except urllib.error.HTTPError as error:
                    actual, data = error.code, error.read()
                assert actual == status, (path, actual, data.decode()[:1000])
                try:
                    return json.loads(data)
                except json.JSONDecodeError:
                    assert status >= 400, data
                    return {"message": data.decode()}

            def scan(request):
                return call("tests/source-candidates", request)

            def run_request(request, facts, candidate=None):
                return {"version": 1, "input": {"kind": "saved"}, "analysis": request,
                        "fingerprint": facts["fingerprint"], "candidate_id": (candidate or facts["candidates"][0])["id"],
                        "selection": {"kind": "name_filter"}}

</ui:copy>

Source inspection may parse declarations, but must not import the module or create a terminal. Unknown execution modes, traversal and stale scan identities refuse admission.

<ui:copy id="source-tests-http-readonly">
            source = "from pathlib import Path\nPath('executed-on-import').touch()\ndef test_present():\n    pass\n"
            plain = {"path": "test_present.py", "source": source}
            facts = scan(plain)
            assert facts["candidates"][0]["name"] == "test_present"
            assert facts["candidates"][0]["run_scope"] == "file"
            assert not (root / "executed-on-import").exists()
            assert not (root / "test_present.py").exists()
            assert call("terminals")["sessions"] == []
            stale = run_request(plain, facts)
            stale["fingerprint"] = "0" * 64
            call("tests/run-source", stale, 409)
            unsupported = run_request(plain, facts)
            unsupported["input"] = {"kind": "captured"}
            call("tests/run-source", unsupported, 422)
            call("tests/source-candidates", {"path": "../outside.py", "source": source}, 422)

</ui:copy>

Compiler projections retain the foreign authored page. Matching generated bytes are insufficient when that page has unsaved changes. Explicit capture registers a memory-only context.

<ui:copy id="source-tests-http-contributors">
            entry = '<h:doc xmlns:h="http://www.hickorydocs.com/1.0"><h:project sources="cases.md"/></h:doc>'
            saved = '<h:doc xmlns:h="http://www.hickorydocs.com/1.0">\n<h:copy id="cases">\ndef test_saved():\n    pass\n</h:copy>\n<h:file path="test_case.py"><h:paste select="#cases"/></h:file>\n</h:doc>'
            draft = saved.replace("test_saved", "test_draft")
            (root / "entry.md").write_text(entry)
            (root / "cases.md").write_text(saved)
            project = {"path": "cases.md", "source": draft, "project_entry": "entry.md", "sources": {"unused.md": "unrelated"}, "capture": True}
            projected = scan(project)
            candidate = projected["candidates"][0]
            assert candidate["source"]["path"] == "cases.md"
            assert candidate["name"] == "test_draft"
            assert projected["unused_sources"] == ["unused.md"]
            assert projected["project_context"]["materialized"] is False
            assert not (root / "test_case.py").exists()
            # Supply matching generated bytes, but retain the stale authored contributor.
            (root / "test_case.py").write_text("def test_draft():\n    pass\n")
            call("tests/run-source", run_request(project, projected), 409)
            assert call("terminals")["sessions"] == []

</ui:copy>

Parameterized source declarations cannot identify concrete runner cases. A name-filter request must fail before any runner is planned.

            assert dynamic_facts["candidates"][0]["name_filter"] is None

Run a dependency-free Rust crate through the real terminal service, then change the saved test and prove that the old selection is refused. Poll only this disposable focused run.

<ui:copy id="source-tests-http-execution">
            # A dependency-free real Rust crate proves execution, not merely command planning.
            (root / "Cargo.toml").write_text('[package]\nname="source_test_fixture"\nversion="0.1.0"\nedition="2024"\n')
            (root / "src").mkdir()
            rust = "#[test]\nfn source_witness() { assert_eq!(2 + 2, 4); }\n"
            (root / "src/lib.rs").write_text(rust)
            request = {"path": "src/lib.rs", "source": rust}
            rust_facts = scan(request)
            assert rust_facts["candidates"][0]["run_scope"] == "crate"
            admitted = call("tests/run-source", run_request(request, rust_facts))
            assert admitted["source_checked_at_admission"] is True
            assert admitted["source_frozen"] is False
            assert admitted["checked_sources"] == rust_facts["source_hashes"]
            session = admitted["id"]
            deadline = time.monotonic() + 40
            while time.monotonic() < deadline:
                output = call(f"terminals/{session}/output?view=screen&offset=0&limit=16384")
                text = bytes(output["bytes"]).decode(errors="replace")
                if "test result: ok." in text and "source_witness ... ok" in text:
                    break
                time.sleep(.1)
            else:
                raise AssertionError(text)
            (root / "src/lib.rs").write_text(rust + "// saved after discovery\n")
            call("tests/run-source", run_request(request, rust_facts), 409)
            assert len(call("terminals")["sessions"]) == 1
            print("PASS source candidates: read-only, compiler origins, contributor guards, explicit scope, real Rust execution")
        finally:
            fixture.stop(engine)


if __name__ == "__main__":
    main()
