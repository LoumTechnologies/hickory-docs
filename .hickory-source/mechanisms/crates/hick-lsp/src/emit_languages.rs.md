//! Emit the editor's language data from the server's routing table.
use hick_lsp::lang_detect::{extensions, file_names, highlight, known_language_ids};

fn main() {
    print!(
        r#"// GENERATED by `just codegen` from crates/hick-plugins language contributions.
// Do not edit: run `just codegen` instead. `just check-codegen` fails on drift.
//
// The server's routing table is the only statement of which languages exist.
// `highlight` says whether a CodeMirror grammar is expected for the id; the
// binding itself lives in `../languages.ts`, and a test there fails if a
// language claims a grammar it has not bound.

"#
    );
    let ids = known_language_ids();
    println!("export type RoutedLanguageId =");
    for id in &ids {
        println!("  | \"{id}\"");
    }
    println!(";\n");
    println!("/** Every routed language id, sorted. */");
    println!("export const ROUTED_LANGUAGE_IDS: readonly RoutedLanguageId[] = [");
    for id in &ids {
        println!("  \"{id}\",");
    }
    println!("];\n");
    println!("/** File extension (lowercase, no dot) to language id. */");
    println!("export const EXTENSION_TO_LANGUAGE: Readonly<Record<string, RoutedLanguageId>> = {{");
    for (ext, id, _) in extensions() {
        println!("  \"{ext}\": \"{id}\",");
    }
    println!("}};\n");
    println!("/** Exact basenames that select a language without an extension. */");
    println!("export const FILE_NAME_TO_LANGUAGE: Readonly<Record<string, RoutedLanguageId>> = {{");
    for (name, id, _) in file_names() {
        println!("  \"{}\": \"{id}\",", name.to_ascii_lowercase());
    }
    println!("}};\n");
    println!("/** Languages the editor is expected to draw with a grammar. */");
    println!("export const HIGHLIGHTED_LANGUAGES: readonly RoutedLanguageId[] = [");
    for id in &ids {
        if highlight(id).is_editor() {
            println!("  \"{id}\",");
        }
    }
    println!("];");
    println!("\n/** Debug targets declared by the language plugins. */");
    println!("export const DEBUGGABLE_LANGUAGE_IDS: readonly string[] = [");
    let mut debuggable: Vec<_> = hick_plugins::registry()
        .languages(hick_plugins::InterfaceKind::Dap)
        .collect();
    debuggable.sort_unstable();
    for id in debuggable {
        println!("  \"{id}\",");
    }
    println!("];\n");
    println!("/** Declared providers; this does not claim tools are installed or running. */");
    println!(
        "export const LANGUAGE_CAPABILITIES: Readonly<Record<string, readonly string[]>> = {{"
    );
    for id in &ids {
        let kinds: Vec<_> = hick_plugins::registry()
            .plugins()
            .filter(|p| p.languages.contains(id))
            .flat_map(|p| p.interfaces)
            .filter(|i| i.supports(id))
            .map(|i| format!("\"{:?}\"", i.kind()))
            .collect();
        println!("  \"{id}\": [{}],", kinds.join(", "));
    }
    println!("}};");
}
