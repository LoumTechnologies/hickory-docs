### `capture`

//! Shared exact-byte mappings for coverage, profiling and source navigation.
use crate::coverage::safe_path;
use anyhow::Result;
use serde_json::{Value, json};
use sha2::{Digest, Sha256};
use std::{collections::BTreeMap, path::Path};
pub fn capture(
    root: &Path,
    hashes: &BTreeMap<String, String>,
    locations: &BTreeMap<String, Vec<u32>>,
) -> Result<BTreeMap<String, BTreeMap<u32, Value>>> {
    let mut mappings = BTreeMap::<String, BTreeMap<u32, Value>>::new();
    for (doc, hash) in hashes {
        if !doc.ends_with(".md") {
            continue;
        }
        let Ok(path) = safe_path(root, doc) else {
            continue;
        };
        let Ok(source) = std::fs::read_to_string(&path) else {
            continue;
        };
        if format!("{:x}", Sha256::digest(source.as_bytes())) != *hash
            || !source.contains("<hick:file")
        {
            continue;
        }
        let Ok(state) = hick_lsp::document::HickDocumentState::from_source(&source) else {
            continue;
        };
        // Map only when the virtual generated bytes match the measured source exactly.
        let matching: Vec<_> = state
            .virtual_files
            .iter()
            .filter_map(|f| {
                let generated = path.parent()?.join(&f.path).canonicalize().ok()?;
                let relative = generated
                    .strip_prefix(root)
                    .ok()?
                    .to_string_lossy()
                    .replace('\\', "/");
                let measured = locations.get(&relative)?;
                let bytes = std::fs::read(&generated).ok()?;
                (bytes == f.written_content().as_bytes()
                    && hashes.get(&relative) == Some(&format!("{:x}", Sha256::digest(&bytes))))
                .then_some((relative, generated, measured.clone()))
            })
            .collect();
        let map = hick_dap::Mapping::for_state(&path, &state, path.parent().unwrap())?;
        for (relative, generated, lines) in matching {
            for line in lines {
                if let Some((source, n)) = map.source_location(&generated, line - 1)
                    && let Ok(source) = source.canonicalize()
                    && source.starts_with(root)
                {
                    mappings.entry(relative.clone()).or_default().insert(
                        line,
                        json!({"path":source.strip_prefix(root)?.to_string_lossy().replace('\\',"/"),"line":n+1}),
                    );
                }
            }
        }
    }
    Ok(mappings)
}
